#include <iostream>
#include <cstdlib>
#include <ctime>

using namespace std;

int main() {
    srand(time(0));

    int number = rand() % 100 + 1;
    int guess;

    cout << "Guess a number from 1 to 100: ";
    cin >> guess;

    while (guess != number) {
        if (guess < number) {
            cout << "Too low! Guess again: ";
        } else {
            cout << "Too high! Guess again: ";
        }

        cin >> guess;
    }

    cout << "You got it!" << endl;

    return 0;
}
